let peer, conn; let isHost = false; // 1. Peer Başlatma ve ID Oluşturma function initPeer() { // Kendi benzersiz Peer ID'mizi oluşturuyoruz peer = new Peer(); peer.on('open', (id) => { console.log("Benim Peer ID'im: " + id); document.getElementById('my-id').innerText = "ID: " + id; }); // Başka biri bize bağlandığında (Host biziz) peer.on('connection', (connection) => { conn = connection; isHost = true; setupConnectionEvents(); console.log("Bir oyuncu bağlandı!"); }); } // 2. Başka Birinin ID'sine Bağlanma (Misafir) function connectToPeer(friendId) { conn = peer.connect(friendId); isHost = false; setupConnectionEvents(); } // 3. Veri Alışverişi ve Senkronizasyon Olayları function setupConnectionEvents() { conn.on('open', () => { console.log("PeerJS Bağlantısı Başarıyla Kuruldu!"); document.getElementById('status-bar').innerText = "Multiplayer Aktif!"; }); conn.on('data', (data) => { // Karşı taraftan gelen ham verileri işle (Blok ekleme, kırma, pozisyon vb.) if (data.type === 'addBlock') { addBlockToScene(data.x, data.y, data.z, data.blockType, false); } else if (data.type === 'removeBlock') { removeBlockFromScene(data.index, false); } else if (data.type === 'playerMove') { // Eğer karşı oyuncunun hareketini senkronize ediyorsan updateOtherPlayerPosition(data.x, data.y, data.z); } }); conn.on('close', () => { alert("Bağlantı kapandı!"); }); } // 4. Veri Gönderme Fonksiyonu (Blok kırıldığında veya konulduğunda) function sendActionToPeer(actionData) { if (conn && conn.open) { conn.send(actionData); } }
